cmake_minimum_required(VERSION 3.22.1)
project(gluco_llm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/CMakeLists.txt")
    message(FATAL_ERROR "No llama.cpp checkout. GitHub Actions checks out pinned upstream commit first.")
endif()
add_subdirectory(llama.cpp ${CMAKE_CURRENT_BINARY_DIR}/llama-build EXCLUDE_FROM_ALL)
add_library(gluco_llm SHARED local_llm.cpp)
target_include_directories(gluco_llm PRIVATE llama.cpp/include)
target_link_libraries(gluco_llm PRIVATE llama log android)
